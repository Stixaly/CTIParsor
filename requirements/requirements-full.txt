# Everything the image installs: the light runtime, then the ML models, OCR,
# the other LLM providers and URL capture.  `make lock` resolves this file
# into requirements.lock.txt (and torch into requirements-torch.lock.txt).
-r requirements.txt

# ── Document ingestion ────────────────────────────────────────────────────────
markitdown>=0.1.0            # PDF / DOCX → structured Markdown
pdf2image>=1.17.0            # PDF → images (scanned PDFs via OCR)
pytesseract>=0.3.10          # Tesseract OCR wrapper for scanned PDFs

# ── NLP / NER ────────────────────────────────────────────────────────────────
# sentence-transformers installs PyTorch (CPU) automatically — ~800 MB first run.
# Required for Stage 2c (semantic TTP detection) and Stage 2d (CyNER).
# Upper bound: each major of sentence-transformers (3→4→5) changed encode() return types.
# Bump the cap intentionally after running: make test-fast
# Floor: 5.6.0 fixes its two OSV advisories.
sentence-transformers>=5.6.0,<6
# Upper bound: HuggingFace Transformers 6.x may drop APIs used in Stage 2d.
# Floor: 5.10.0 is the last fix among the 50 OSV advisories on 4.35–5.9.
transformers>=5.10.0,<6      # HuggingFace Transformers (CyNER model, Stage 2d)
# CyNER 2.0 is a deberta-v2 model — its tokenizer requires SentencePiece.
sentencepiece>=0.2.1         # Tokenizer backend for CyNER 2.0 (DeBERTa-v3, Stage 2d); 0.2.1 fixes PYSEC-2026-1909

# ── Zero-shot NER (GLiNER — Stage 2e) ────────────────────────────────────────
# Generalist and Lightweight NER model — zero-shot, no fine-tuning needed.
# Paper: "0-CTI: A Modular Framework for Zero-Shot CTI Extraction" (ADR-004 P2-A)
# CPU-friendly (~300 MB). Discovers entity types CyNER cannot (sectors, campaigns…)
gliner>=0.2.0

# spaCy: the legacy NER fallback of Stage 2, imported with try/except — the
# pipeline works without it (gazetteer + CyNER + GLiNER cover NER).  No model
# is downloaded by default:
#   python -m spacy download en_core_web_sm   (fast, lower accuracy)
#   python -m spacy download en_core_web_lg   (recommended)
spacy>=3.7.0,<4

# ── LLM providers ────────────────────────────────────────────────────────────
# Upper bound: openai 2.x introduced breaking changes from 1.x (Mistral/Ollama client).
# Bump the cap intentionally after running: make test-fast
openai>=1.30.0,<3            # Mistral AI / Ollama / vLLM (OpenAI-compatible endpoint)

# ── URL capture (ADR-0029, the "URL" tab) ─────────────────────────────────────
# The image installs Chromium and its system libraries.  On a host:
#   pip install playwright
#   python -m playwright install chromium
#   sudo python -m playwright install-deps chromium   <- needs root
# The last step is the one people miss: without those libraries Chromium is
# installed but dies at launch (libasound.so.2).  The API degrades cleanly:
# /api/ingest/url answers 503 and the UI asks for the report text instead.
playwright>=1.44
